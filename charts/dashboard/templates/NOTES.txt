Watch it come up:
  kubectl -n {{ .Release.Namespace }} get pods -o wide -w

  curl -s http://{{ .Values.nfs.server }}/api/health

  helm upgrade {{ .Release.Name }} <chart> -n {{ .Release.Namespace }} --reuse-values --set backend.replicas=5
